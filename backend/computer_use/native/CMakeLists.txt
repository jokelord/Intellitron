cmake_minimum_required(VERSION 3.16)
project(computer_use_native C)
set(CMAKE_C_STANDARD 11)
include(CTest)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO_PIPEWIRE QUIET IMPORTED_TARGET gio-unix-2.0 libpipewire-0.3 libspa-0.2)
pkg_check_modules(WAYLAND_INPUT QUIET IMPORTED_TARGET json-glib-1.0 libei-1.0)
pkg_check_modules(X11_INPUT QUIET IMPORTED_TARGET x11 xext xtst xrandr)
pkg_check_modules(JSON_GLIB QUIET IMPORTED_TARGET json-glib-1.0)
add_executable(computer-use-helper main.c ipc.c control.c wayland_probe.c portal.c capture.c ei_input.c x11.c live.c)
string(TOLOWER "${CMAKE_SYSTEM_NAME}" COMPUTER_USE_INSTALL_OS)
install(TARGETS computer-use-helper
  RUNTIME DESTINATION "bin/${COMPUTER_USE_INSTALL_OS}-${CMAKE_SYSTEM_PROCESSOR}"
  COMPONENT computer-use-helper)
target_compile_options(computer-use-helper PRIVATE -Wall -Wextra -Werror)
if(JSON_GLIB_FOUND)
  target_compile_definitions(computer-use-helper PRIVATE HAVE_JSON_GLIB=1)
  target_link_libraries(computer-use-helper PRIVATE PkgConfig::JSON_GLIB)
endif()
if(GIO_PIPEWIRE_FOUND)
  target_compile_definitions(computer-use-helper PRIVATE HAVE_GIO_PIPEWIRE_DEPS=1)
  target_link_libraries(computer-use-helper PRIVATE PkgConfig::GIO_PIPEWIRE)
else()
  message(WARNING "GIO/PipeWire development dependencies incomplete; Portal capture remains unavailable.")
endif()
if(WAYLAND_INPUT_FOUND)
  target_compile_definitions(computer-use-helper PRIVATE HAVE_WAYLAND_INPUT_DEPS=1)
  target_link_libraries(computer-use-helper PRIVATE PkgConfig::WAYLAND_INPUT)
else()
  message(WARNING "json-glib/libei development dependencies incomplete; Wayland input remains unavailable.")
endif()
if(X11_INPUT_FOUND)
  target_compile_definitions(computer-use-helper PRIVATE HAVE_X11_INPUT_DEPS=1)
  target_link_libraries(computer-use-helper PRIVATE PkgConfig::X11_INPUT m)
else()
  message(WARNING "X11 development dependencies incomplete; X11 capture/input remains unavailable.")
endif()
if(GIO_PIPEWIRE_FOUND AND WAYLAND_INPUT_FOUND AND X11_INPUT_FOUND)
  target_compile_definitions(computer-use-helper PRIVATE HAVE_PLATFORM_DEPS=1)
endif()
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND AND JSON_GLIB_FOUND AND X11_INPUT_FOUND)
    add_test(NAME native_x11_protocol COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/native_computer_use/test_x11_protocol.py
      $<TARGET_FILE:computer-use-helper>)
    set_tests_properties(native_x11_protocol PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
  endif()
  if(JSON_GLIB_FOUND AND GIO_PIPEWIRE_FOUND)

    set_tests_properties(native_control_parser PROPERTIES TIMEOUT 15)
  endif()
  if(JSON_GLIB_FOUND AND GIO_PIPEWIRE_FOUND AND WAYLAND_INPUT_FOUND)
    add_executable(test-live-commands ../../../tests/native_computer_use/test_live_commands.c live.c portal.c capture.c ei_input.c x11.c)

  if(WAYLAND_INPUT_FOUND)
    add_executable(test-ei-policy ../../../tests/native_computer_use/test_ei_policy.c ei_input.c)

    add_executable(test-capture-negotiation ../../../tests/native_computer_use/test_capture_negotiation.c)
    target_include_directories(test-capture-negotiation PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_definitions(test-capture-negotiation PRIVATE HAVE_GIO_PIPEWIRE_DEPS=1)
    target_compile_options(test-capture-negotiation PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    target_link_libraries(test-capture-negotiation PRIVATE PkgConfig::GIO_PIPEWIRE)
    add_test(NAME native_capture_negotiation COMMAND test-capture-negotiation)
    add_executable(test-portal ../../../tests/native_computer_use/test_portal.c portal.c)
    target_include_directories(test-portal PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_definitions(test-portal PRIVATE HAVE_GIO_PIPEWIRE_DEPS=1)
    target_compile_options(test-portal PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    target_link_libraries(test-portal PRIVATE PkgConfig::GIO_PIPEWIRE)
    add_test(NAME native_portal_result COMMAND test-portal)
  endif()
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND AND GIO_PIPEWIRE_FOUND)
    add_test(NAME native_portal_lifecycle COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/native_computer_use/run_private_portal.py $<TARGET_FILE:test-portal>)
    set_tests_properties(native_portal_lifecycle PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 15)
  endif()
  if(Python3_FOUND AND JSON_GLIB_FOUND AND GIO_PIPEWIRE_FOUND)
    add_test(NAME native_control_protocol COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/native_computer_use/test_control.py
      $<TARGET_FILE:computer-use-helper>)
    set_tests_properties(native_control_protocol PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)

  add_executable(test-live-buffers ../../../tests/native_computer_use/test_live_buffers.c capture.c)
  target_include_directories(test-live-buffers PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
